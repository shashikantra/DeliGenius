DATA INGESTION

--Define your paths
$ArchiveDir = "C:\path\to\your\archive"
$InboxDir = "C:\Shashi\MyProjects\DeliGenius\samples\inbox"

--Clear the inbox
Remove-Item -Path "$InboxDir\*" -Force

-- Move archived files to the inbox
Copy-Item -Path "$ArchiveDir\*" -Destination $InboxDir -Recurse -Force

--Verify the move
Get-ChildItem -Path $InboxDir | Select-Object Name

--Run the ingest
deli-rag ingest samples/inbox

🖥️ Terminal 3: Frontend (React / Vite)
Purpose: Runs the user interface in your browser.

powershell
# 1. Go to the frontend folder (no need to activate .venv here)
cd C:\Shashi\MyProjects\DeliGenius\frontend

# 2. Install the node dependencies (you only need to do this the first time)
npm install

# 3. Start the Vite dev server
npm run dev
Wait until it says Local: http://localhost:5173/.
